cmake_minimum_required(VERSION 3.25)
project(BattleCore VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
file(GLOB CORE CONFIGURE_DEPENDS src/*.cpp)
if(CORE)
 add_library(battle_core STATIC ${CORE})
 target_include_directories(battle_core PUBLIC include)
 target_compile_features(battle_core PUBLIC cxx_std_20)
 if(MSVC)
  target_compile_options(battle_core PRIVATE /W4 /GR- /EHs-c-)
 else()
  target_compile_options(battle_core PRIVATE -Wall -Wextra -Wpedantic -fno-rtti -fno-exceptions -ffp-contract=off)
 endif()
else()
 add_library(battle_core INTERFACE)
 target_include_directories(battle_core INTERFACE include)
 target_compile_features(battle_core INTERFACE cxx_std_20)
endif()
if(BUILD_TESTING)
 file(GLOB TESTS CONFIGURE_DEPENDS tests/*Tests.cpp)
 add_executable(battle_tests tests/TestMain.cpp ${TESTS})
 target_include_directories(battle_tests PRIVATE third_party tools tests)
 target_link_libraries(battle_tests PRIVATE battle_core)
 if(MINGW)
  target_link_options(battle_tests PRIVATE -static)
 endif()
 target_compile_definitions(battle_tests PRIVATE BATTLE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 foreach(pair world:World import:Import formation:Formation navigation:Navigation officers:Officer combat:Combat control:Control replay:Replay artifacts:Artifact batch:Batch integration:Integration)
  string(REPLACE ":" ";" parts ${pair})
  list(GET parts 0 group)
  list(GET parts 1 file)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${file}Tests.cpp")
   add_test(NAME battle.${group} COMMAND battle_tests --test-case=${group}.* --no-skip)
  endif()
 endforeach()
endif()
